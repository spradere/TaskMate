# get repo size
git count-objects -vH

# search in commits, without merge, most add
git log --no-merges --numstat --date=iso-strict --pretty="C %H %ad %s" \
| awk '
  /^[0-9]/ { add += $1; next }
  /^C / {
    if (NR > 1) print add, prev
    prev = $0
    add = 0
  }
  END { print add, prev }
' \
| sort -k1,1nr -k3,3r \
| head -20

# big blop
git rev-list --objects --all \
| git cat-file --batch-check='%(objecttype) %(objectname) %(objectsize) %(rest)' \
| awk '$1=="blob"{print $3, $2, $4}' \
| sort -n \
| tail -20
